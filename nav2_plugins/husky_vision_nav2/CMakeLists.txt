cmake_minimum_required(VERSION 3.8)
project(husky_vision_nav2)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_library(vision_mask_layer SHARED src/vision_mask_layer.cpp)
target_include_directories(vision_mask_layer PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(vision_mask_layer nav2_costmap_2d nav_msgs pluginlib rclcpp tf2 tf2_ros tf2_geometry_msgs)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS vision_mask_layer EXPORT export_vision_mask_layer LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_vision_mask_layer HAS_LIBRARY_TARGET)
ament_export_dependencies(nav2_costmap_2d nav_msgs pluginlib rclcpp tf2 tf2_ros tf2_geometry_msgs)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest QUIET)
  if(ament_cmake_gtest_FOUND)
    ament_add_gtest(test_vision_mask_layer test/test_vision_mask_layer.cpp)
    target_link_libraries(test_vision_mask_layer vision_mask_layer)
    ament_target_dependencies(test_vision_mask_layer nav2_costmap_2d nav_msgs rclcpp tf2_ros)
  endif()
endif()
ament_package()
